# Fondamentaux Cryptographiques

## Chiffrement symétrique

```bash
# OpenSSL : chiffrement AES-256-GCM
# Chiffrer un fichier
openssl enc -aes-256-gcm -pbkdf2 -iter 100000 \
  -in fichier.txt -out fichier.enc -k MonMotDePasse

# Déchiffrer
openssl enc -d -aes-256-gcm -pbkdf2 -iter 100000 \
  -in fichier.enc -out fichier.txt -k MonMotDePasse

# Chiffrement avec clé dérivée (PBKDF2)
openssl enc -aes-256-cbc -md sha256 -pbkdf2 -salt \
  -in secret.txt -out secret.enc

# Générer une clé aléatoire (32 octets = 256 bits)
openssl rand -hex 32
openssl rand -base64 32
```

## Chiffrement asymétrique (RSA / EC)

```bash
# Générer une paire RSA 4096 bits
openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:4096 \
  -out private.pem
openssl rsa -in private.pem -pubout -out public.pem

# Générer une paire EC (Curve25519 recommandé)
openssl genpkey -algorithm X25519 -out ec-private.pem
openssl pkey -in ec-private.pem -pubout -out ec-public.pem

# Chiffrement RSA (asymm)
openssl pkeyutl -encrypt -pubin -inkey public.pem \
  -in secret.txt -out secret.enc
openssl pkeyutl -decrypt -inkey private.pem \
  -in secret.enc -out secret.txt

# Signature numérique
openssl dgst -sha256 -sign private.pem -out signature.bin document.pdf
openssl dgst -sha256 -verify public.pem -signature signature.bin document.pdf
```

## Hachage et intégrité

```bash
# Hachage de fichiers
sha256sum fichier.iso > fichier.iso.sha256
sha256sum -c fichier.iso.sha256  # Vérification

# Comparaisons rapides
md5sum fichier.txt
sha1sum fichier.txt
sha256sum fichier.txt
sha512sum fichier.txt

# HMAC (intégrité avec clé)
openssl dgst -sha256 -hmac "clé-secrète" fichier.txt

# Hachage de chaîne
echo -n "texte" | sha256sum
echo -n "texte" | openssl dgst -sha256

# Cracker un hash (dictionnaire) avec john
echo '$6$sel$hash_sha512crypt' > hash.txt
john hash.txt --wordlist=/usr/share/wordlists/rockyou.txt
john hash.txt --show
```

## TLS et certificats

```bash
# Inspecter un certificat TLS distant
openssl s_client -connect example.com:443 -showcerts
openssl s_client -connect example.com:443 </dev/null 2>/dev/null | \
  openssl x509 -noout -dates -subject -issuer

# Lire un certificat local
openssl x509 -in cert.pem -noout -text
openssl x509 -in cert.pem -noout -dates    # Validité
openssl x509 -in cert.pem -noout -subject  # Sujet
openssl x509 -in cert.pem -noout -issuer   # Émetteur
opensssl x509 -in cert.pem -noout -fingerprint -sha256  # Empreinte

# Vérifier la chaîne de confiance
openssl verify -CAfile /etc/ssl/certs/ca-certificates.crt cert.pem
```
