```powershell
[CmdletBinding()]
param(
    [Parameter()]
    [string[]]$Cibles = @("SRV-APP01", "SRV-APP02", "SRV-APP03", "SRV-APP04", "SRV-APP05", "SRV-APP06"),

    [Parameter()]
    [int]$DelaiSimule = 2,

    [Parameter()]
    [switch]$ModeSynchrone
)

$script:Resultats = @()

$Chrono = Measure-Command {
    if ($ModeSynchrone) {

        Write-Host "sequentiel sélectionné." 

        $script:Resultats = foreach ($Machine in $Cibles) {
            Write-Host "Traitement de $Machine"
            Start-Sleep -Seconds $DelaiSimule

            [PSCustomObject]@{
                Machine       = $Machine
                PID_Execution = $PID
                Statut        = "OK"
                HeureFin      = (Get-Date -Format "HH:mm:ss")
            }
        }
    }
    else {

        Write-Host "PARALLÈLE sélectionné" -ForegroundColor Cyan
        Write-Host "Lancement de $($Cibles.Count) jobs d'arrière-plan"

        $Jobs = foreach ($Machine in $Cibles) {
            Start-Job -Name "Job_$Machine" -ScriptBlock {
                param($Nom, $Delai)

                Start-Sleep -Seconds $Delai

                [PSCustomObject]@{
                    Machine       = $Nom
                    PID_Execution = $PID
                    Statut        = "OK"
                    HeureFin      = (Get-Date -Format "HH:mm:ss")
                }
            } -ArgumentList $Machine, $DelaiSimule
        }

        try {
            while (($Jobs | Where-Object State -eq 'Running').Count -gt 0) {
                $Termines = ($Jobs | Where-Object State -ne 'Running').Count
                $Total    = $Jobs.Count
                $Ratio    = [math]::Round(($Termines / $Total) * 100)

                Write-Progress -Activity "Exécution des tâches d'audit" `
                               -Status "$Termines / $Total tâches achevées" `
                               -PercentComplete $Ratio

                Start-Sleep -Milliseconds 500
            }
            Write-Progress -Activity "Exécution des tâches d'audit" -Completed

            $script:Resultats = $Jobs | Receive-Job
        }
        finally {
            $Jobs | Remove-Job -Force
        }
    }
}

$DureeTotale = [math]::Round($Chrono.TotalSeconds, 2)


if ($ModeSynchrone) {
    Write-Host "`n[FIN] Durée totale constatée : $DureeTotale secondes." -ForegroundColor Yellow
}
else {
    Write-Host ""
    $script:Resultats | Select-Object -Property Machine, PID_Execution, Statut, HeureFin | Format-Table -AutoSize


    $DureeTheorique = $Cibles.Count * $DelaiSimule
    $GainPct = [math]::Round(((($DureeTheorique - $DureeTotale) / $DureeTheorique) * 100))

    Write-Host "Traitement fini en $DureeTotale secondes (Gain constaté : ~$GainPct %)." -ForegroundColor Green
}
```
