```powershell
[CmdletBinding()]
param(
    [string[]]$ServeursCibles = @($env:COMPUTERNAME, "localhost"),
    [int]$SeuilRAMMo = 150,
    [switch]$ExporterCSV
)

$DossierAudit = "C:\Audit"
$FichierCSV   = "$DossierAudit\audit_remoting.csv"

$ServeursValides = @()

foreach ($Serveur in $ServeursCibles) {
    if (Test-WSMan -ComputerName $Serveur -ErrorAction SilentlyContinue) {
        Write-Host "Serveur $Serveur : Joignable"
        $ServeursValides += $Serveur
    }
    else {
        Write-Host "Serveur $Serveur : Injoignable (Echec WinRM)"
    }
}

if ($ServeursValides.Count -eq 0) {
    Write-Warning "Aucun serveur cible n'est joignable via WinRM. Arret de l'audit."
    return
}


$Resultats = Invoke-Command -ComputerName $ServeursValides -ScriptBlock {
    Get-Process |
        Where-Object { $_.WorkingSet64 -gt ($using:SeuilRAMMo * 1MB) } |
        Select-Object @{Name = "Machine"; Expression = { $env:COMPUTERNAME }},
                      ProcessName,
                      Id,
                      @{Name = "RAM_Mo"; Expression = { [math]::Round($_.WorkingSet64 / 1MB) }}
} -ErrorAction SilentlyContinue

$DonneesFormatees = $Resultats | Select-Object Machine, ProcessName, Id, RAM_Mo | Sort-Object Machine, RAM_Mo -Descending


$DonneesFormatees | Format-Table -AutoSize

if ($ExporterCSV) {
    if (-not (Test-Path -Path $DossierAudit)) {
        New-Item -Path $DossierAudit -ItemType Directory -Force | Out-Null
    }
    $DonneesFormatees | Export-Csv -Path $FichierCSV -NoTypeInformation -Delimiter ";" -Encoding UTF8
    Write-Host "`nRapport exporte avec succes : $FichierCSV" -ForegroundColor Green
}

Write-Host "`nAudit termine."
```
