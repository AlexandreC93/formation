```powershell
[CmdletBinding()]
param(
    [Parameter(Mandatory = $false)]
    [string]$CheminCle = "HKLM:\SOFTWARE\AscentCorporate",

    [Parameter(Mandatory = $false)]
    [string]$NomPropriete = "SiteCode",

    [Parameter(Mandatory = $false)]
    [string]$ValeurAttendue = "PARIS-01",

    [switch]$Corriger
)

Write-Host "Cle analysee       : $CheminCle"
Write-Host "Propriete cible    : $NomPropriete"
Write-Host "Valeur de reference : $ValeurAttendue"
Write-Host "Mode remediation   : $(if ($Corriger) { 'ACTIF' } else { 'DESACTIF (Audit seul)' })"

try {
    $CleExiste = Test-Path -Path $CheminCle

    if (-not $CleExiste) {
        Write-Host "[NON CONFORME] La cle de registre '$CheminCle' est introuvable." -ForegroundColor Red

        if ($Corriger) {
            Write-Host "[ACTION] Creation de la cle '$CheminCle'..." -ForegroundColor Yellow
            New-Item -Path $CheminCle -ItemType Directory -Force -ErrorAction Stop | Out-Null
            Write-Host "[OK] Cle creee avec succes." -ForegroundColor Green
        }
        else {
            Write-Warning "Aucune correction appliquee. Relancez avec le commutateur -Corriger."
            return
        }
    }
    else {
        Write-Host "[OK] La cle de registre existe." -ForegroundColor Green
    }

    # 2. Verification de la propriete (l'entree dans la cle)
    $ObjetRegistre = Get-ItemProperty -Path $CheminCle -Name $NomPropriete -ErrorAction SilentlyContinue

    if ($null -eq $ObjetRegistre -or $null -eq $ObjetRegistre.$NomPropriete) {
        Write-Host "[NON CONFORME] La propriete '$NomPropriete' n'existe pas dans la cle." -ForegroundColor Red

        if ($Corriger) {
            Write-Host "[ACTION] Creation et initialisation de la propriete '$NomPropriete'..." -ForegroundColor Yellow
            New-ItemProperty -Path $CheminCle -Name $NomPropriete -Value $ValeurAttendue -PropertyType String -Force -ErrorAction Stop | Out-Null
            Write-Host "[CORRIGE] Propriete definie sur '$ValeurAttendue'." -ForegroundColor Green
        }
        else {
            Write-Warning "La valeur est absente. Utilisez -Corriger pour la deployer."
        }
    }
    else {
        $ValeurActuelle = $ObjetRegistre.$NomPropriete
        Write-Host "Valeur actuellement configuree : '$ValeurActuelle'" -ForegroundColor Gray

        # 3. Verification de la conformite de la valeur
        if ($ValeurActuelle -eq $ValeurAttendue) {
            Write-Host "[CONFORME] La valeur est exactement conforme aux exigences de securite." -ForegroundColor Green
        }
        else {
            Write-Host "[NON CONFORME] Valeur non conforme detectee (attendue : '$ValeurAttendue')." -ForegroundColor Red

            if ($Corriger) {
                Write-Host "[ACTION] Mise a jour de la valeur..." -ForegroundColor Yellow
                Set-ItemProperty -Path $CheminCle -Name $NomPropriete -Value $ValeurAttendue -ErrorAction Stop
                Write-Host "[CORRIGE] Valeur modifiee avec succes en '$ValeurAttendue'." -ForegroundColor Green
            }
            else {
                Write-Warning "Valeur divergente. Relancez avec -Corriger pour forcer la mise en conformite."
            }
        }
    }
}
catch [System.UnauthorizedAccessException] {
    Write-Host "`n[ERREUR CRITIQUE] Acces refuse. L'ecriture dans HKLM requiert d'executer PowerShell en tant qu'administrateur." -ForegroundColor Red
}
catch {
    Write-Host "`n[ERREUR CRITIQUE] Une exception inattendue est survenue : $($_.Exception.Message)" -ForegroundColor Red
}
finally {
    Write-Host "`nTraitement termine." -ForegroundColor Cyan
}
```
