# Corrigé — Infrastructure PKI

## Exercice 1

```bash
mkdir -p /opt/pki/{root-ca,intermediate-ca,certs}

# Root CA (hors ligne en production)
openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:4096 -out /opt/pki/root-ca/ca.key
chmod 400 /opt/pki/root-ca/ca.key
openssl req -new -x509 -days 3650 -key /opt/pki/root-ca/ca.key \
  -out /opt/pki/root-ca/ca.crt \
  -subj "/C=FR/ST=IDF/O=Formation/OU=PKI/CN=Formation Root CA"

# Intermediate CA
openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:4096 \
  -out /opt/pki/intermediate-ca/int.key
chmod 400 /opt/pki/intermediate-ca/int.key
openssl req -new -key /opt/pki/intermediate-ca/int.key \
  -out /opt/pki/intermediate-ca/int.csr \
  -subj "/C=FR/ST=IDF/O=Formation/CN=Formation Intermediate CA"

cat > /tmp/v3_ca.ext << 'EOF'
basicConstraints = critical, CA:TRUE, pathlen:0
keyUsage = critical, keyCertSign, cRLSign
subjectKeyIdentifier = hash
authorityKeyIdentifier = keyid:always
EOF

openssl x509 -req -days 1825 -in /opt/pki/intermediate-ca/int.csr \
  -CA /opt/pki/root-ca/ca.crt -CAkey /opt/pki/root-ca/ca.key \
  -CAcreateserial -out /opt/pki/intermediate-ca/int.crt \
  -extfile /tmp/v3_ca.ext

# Certificat serveur avec SAN
cat > /tmp/san-formation.ext << 'EOF'
subjectAltName = DNS:formation.local, DNS:www.formation.local, IP:127.0.0.1
keyUsage = critical, digitalSignature, keyEncipherment
extendedKeyUsage = serverAuth
EOF

openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:2048 \
  -out /opt/pki/certs/server.key
openssl req -new -key /opt/pki/certs/server.key \
  -out /opt/pki/certs/server.csr \
  -subj "/CN=formation.local"
openssl x509 -req -days 365 -in /opt/pki/certs/server.csr \
  -CA /opt/pki/intermediate-ca/int.crt -CAkey /opt/pki/intermediate-ca/int.key \
  -CAcreateserial -out /opt/pki/certs/server.crt -extfile /tmp/san-formation.ext

# Chaîne et vérification
cat /opt/pki/intermediate-ca/int.crt /opt/pki/root-ca/ca.crt > /opt/pki/chain.crt
openssl verify -CAfile /opt/pki/chain.crt /opt/pki/certs/server.crt
```

## Exercice 2 — mTLS

```bash
# Certificat client
openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:2048 \
  -out /opt/pki/certs/client-stagiaire01.key
openssl req -new -key /opt/pki/certs/client-stagiaire01.key \
  -out /opt/pki/certs/client-stagiaire01.csr \
  -subj "/CN=stagiaire01/emailAddress=stagiaire01@formation.local"
openssl x509 -req -days 365 \
  -in /opt/pki/certs/client-stagiaire01.csr \
  -CA /opt/pki/intermediate-ca/int.crt \
  -CAkey /opt/pki/intermediate-ca/int.key \
  -CAcreateserial -out /opt/pki/certs/client-stagiaire01.crt

# Config nginx mTLS
cat > /etc/nginx/sites-available/mtls << 'EOF'
server {
    listen 443 ssl;
    ssl_certificate     /opt/pki/certs/server.crt;
    ssl_certificate_key /opt/pki/certs/server.key;
    ssl_client_certificate /opt/pki/chain.crt;
    ssl_verify_client on;
    ssl_protocols TLSv1.3 TLSv1.2;
    ssl_ciphers 'ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384';
    root /var/www/html;
}
EOF

# Test avec curl
curl --cacert /opt/pki/chain.crt \
  --cert /opt/pki/certs/client-stagiaire01.crt \
  --key /opt/pki/certs/client-stagiaire01.key \
  https://formation.local/

# Sans cert : doit échouer
curl --cacert /opt/pki/chain.crt https://formation.local/
```

## Exercice 3 — CRL

```bash
# Révocation
openssl ca -config /etc/ssl/openssl.cnf \
  -revoke /opt/pki/certs/client-stagiaire01.crt
openssl ca -gencrl -config /etc/ssl/openssl.cnf -out /opt/pki/crl.pem

# Vérifier la révocation
openssl verify -CAfile /opt/pki/chain.crt \
  -crl_check -CRLfile /opt/pki/crl.pem \
  /opt/pki/certs/client-stagiaire01.crt
# Expected: error 23 at 0 depth lookup: certificate revoked
```

## Exercice 4 — OCSP

```bash
# CRL : liste prétéléchargée (risque de délai entre révocation et publication)
# OCSP : requête en temps réel (plus précis, mais dépendance réseau)

# Vérifier OCSP d'un cert Let's Encrypt
openssl s_client -connect example.com:443 -status </dev/null 2>/dev/null | \
  grep -A5 'OCSP Response'

# OCSP Stapling nginx
cat >> /etc/nginx/sites-available/mtls << 'EOF'
# Dans le bloc server :
ssl_stapling on;
ssl_stapling_verify on;
ssl_trusted_certificate /opt/pki/chain.crt;
resolver 8.8.8.8 8.8.4.4 valid=300s;
resolver_timeout 5s;
EOF
nginx -t && systemctl reload nginx
```
