```powershell
[CmdletBinding()]
param(
    [Parameter(Mandatory = $false)]
    [string]$DossierCible = "C:\Windows",

    [Parameter(Mandatory = $false)]
    [int]$SeuilMo = 50,

    [switch]$ModeDetaille
)

$DossierAudit = "C:\Audit"
$FichierLog   = "$DossierAudit\audit.log"

if (-not (Test-Path -Path $DossierAudit)) {
    New-Item -Path $DossierAudit -ItemType Directory -Force | Out-Null
}

function Write-AuditLog {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [string]$Message,

        [Parameter(Mandatory = $false)]
        [ValidateSet("INFO", "WARNING", "ERROR")]
        [string]$Niveau = "INFO"
    )

    $Horodatage = (Get-Date).ToString("yyyy-MM-dd HH:mm:ss")
    $EntreeLog  = "[$Horodatage] [$Niveau] $Message"

    switch ($Niveau) {
        "INFO"    { Write-Host $EntreeLog -ForegroundColor Cyan }
        "WARNING" { Write-Host $EntreeLog -ForegroundColor Yellow }
        "ERROR"   { Write-Host $EntreeLog -ForegroundColor Red }
    }

    $EntreeLog | Add-Content -Path $FichierLog -Encoding UTF8
}

try {
    Write-AuditLog -Message "Demarrage de l'audit sur le dossier '$DossierCible' (seuil : $SeuilMo Mo)" -Niveau INFO

    if (-not (Test-Path -Path $DossierCible)) {
        throw "Le dossier cible '$DossierCible' est introuvable."
    }

    $Fichiers = Get-ChildItem -Path $DossierCible -File -ErrorAction Stop

    $SeuilOctets    = $SeuilMo * 1MB
    $FichiersLourds = $Fichiers | Where-Object { $_.Length -gt $SeuilOctets }

    if ($null -eq $FichiersLourds -or $FichiersLourds.Count -eq 0) {
        Write-AuditLog -Message "Aucun fichier ne depasse le seuil de $SeuilMo Mo." -Niveau INFO
    }
    else {
        Write-AuditLog -Message "$($FichiersLourds.Count) fichier(s) depassant le seuil detecte(s)." -Niveau WARNING

        foreach ($Fichier in $FichiersLourds) {
            $TailleMo = [math]::Round($Fichier.Length / 1MB, 2)
            Write-AuditLog -Message "Fichier volumineux : $($Fichier.Name) ($TailleMo Mo)" -Niveau WARNING
        }
    }

    if ($ModeDetaille) {
        Write-AuditLog -Message "Mode detaille : $($Fichiers.Count) fichier(s) total analyses dans '$DossierCible'." -Niveau INFO
    }
}
catch {
    Write-AuditLog -Message "Echec critique de l'audit : $($_.Exception.Message)" -Niveau ERROR
}
finally {
    Write-AuditLog -Message "Session d'audit terminee. Log disponible dans '$FichierLog'." -Niveau INFO
}
```
