# Gestion des Paquets & Compilation

## APT — Debian/Ubuntu

```bash
# Mettre à jour la liste des paquets
apt update

# Mettre à jour les paquets installés
apt upgrade -y
apt full-upgrade -y   # Peut supprimer des paquets obsolètes

# Installer / supprimer
apt install nginx curl htop -y
apt remove nginx        # Garde la config
apt purge nginx         # Supprime aussi la config
apt autoremove          # Nettoyer les dépendances orphelines

# Recherche
apt search "web server"
apt show nginx
dpkg -l | grep nginx    # Paquets installés

# Audit
dpkg --verify           # Vérifier l'intégrité des paquets
dpkg -L nginx           # Fichiers installés par un paquet
dpkg -S /usr/sbin/nginx # Quel paquet a installé ce fichier
```

## DNF / YUM — RHEL/CentOS/Rocky

```bash
# Mettre à jour
dnf update -y
dnf check-update

# Installer / supprimer
dnf install httpd -y
dnf remove httpd
dnf autoremove

# Recherche
dnf search nginx
dnf info nginx
dnf list installed | grep nginx

# Groupes de paquets
dnf group list
dnf group install "Development Tools"

# Repos
dnf repolist
dnf config-manager --add-repo https://...
```

## Compilation depuis les sources

```bash
# Prérequis : outils de compilation
apt install build-essential gcc make cmake git -y

# Workflow classique (autotools)
wget https://example.com/software-1.0.tar.gz
tar -xzf software-1.0.tar.gz
cd software-1.0/
./configure --prefix=/usr/local --without-debug
make -j$(nproc)          # Compilation parallèle
make check               # Tests unitaires
sudo make install

# Workflow CMake
mkdir build && cd build
cmake .. -DCMAKE_INSTALL_PREFIX=/usr/local -DCMAKE_BUILD_TYPE=Release
make -j$(nproc)
sudo make install

# Créer un paquet .deb depuis les sources
apt install checkinstall
./configure && make
sudo checkinstall         # Crée un .deb installable
```

## Gestion des dépendances et bibliothèques

```bash
# Lister les bibliothèques requises par un exécutable
ldd /usr/sbin/nginx

# Mettre à jour le cache des bibliothèques
ldconfig

# Trouver une bibliothèque
ldconfig -p | grep libssl

# Variables d'environnement des bibliothèques
export LD_LIBRARY_PATH=/usr/local/lib:$LD_LIBRARY_PATH
export PKG_CONFIG_PATH=/usr/local/lib/pkgconfig

# Informations sur un exécutable
file /usr/bin/python3
readelf -d /usr/bin/python3 | grep NEEDED
```

## Signature et vérification des paquets

```bash
# Vérifier la signature GPG d'un paquet Debian
apt-key list
gpg --verify package.deb.sig package.deb

# Sur RHEL : vérifier la signature RPM
rpm --checksig package.rpm
rpm -qa --qf '%{NAME}-%{VERSION}: %{SIGPGP:pgpsig}\n' | head

# Ajouter une clé de dépôt (méthode moderne)
curl -fsSL https://repo.example.com/gpg.key | \
  gpg --dearmor -o /usr/share/keyrings/example.gpg
echo "deb [signed-by=/usr/share/keyrings/example.gpg] https://repo.example.com stable main" \
  > /etc/apt/sources.list.d/example.list
```
