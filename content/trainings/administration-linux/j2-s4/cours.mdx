# Analyse de Trafic & Détection d'Intrusion

## tcpdump — Capture avancée

```bash
# Filtres BPF utiles
tcpdump -i eth0 'tcp and port 443'
tcpdump -i eth0 'host 192.168.1.1'
tcpdump -i eth0 'net 10.0.0.0/8'
tcpdump -i eth0 'not port 22'           # Exclure SSH
tcpdump -i eth0 'tcp[tcpflags] & tcp-syn != 0'  # SYN uniquement
tcpdump -i eth0 'icmp[icmptype]==icmp-echo'     # ICMP pings

# Capture avec décodage
tcpdump -i eth0 -A -s 0 'port 80'      # ASCII
tcpdump -i eth0 -X -s 0 'port 53'      # Hex + ASCII

# Rotation de captures
tcpdump -i eth0 -G 3600 -w /var/log/captures/capture-%Y%m%d-%H.pcap &

# Statistiques
tcpdump -r capture.pcap -qn | \
  awk '{print $3}' | cut -d. -f1-4 | sort | uniq -c | sort -rn | head
```

## Suricata IDS/IPS

```bash
# Installation
apt install suricata -y

# Configuration de base (/etc/suricata/suricata.yaml)
# Interface d'inspection
# af-packet:
#   - interface: eth0

# Mettre à jour les règles (Emerging Threats)
suricata-update
suricata-update list-sources
suricata-update enable-source et/open
suricata-update

# Lancer en mode IDS (surveillance)
suricata -c /etc/suricata/suricata.yaml -i eth0

# Mode offline (analyse de PCAP)
suricata -r /tmp/capture.pcap -c /etc/suricata/suricata.yaml \
  -l /tmp/suricata-output/

# Analyser les alertes
tail -f /var/log/suricata/fast.log
jq '.event_type, .src_ip, .alert.signature' /var/log/suricata/eve.json
```

## Analyse des logs d'authentification

```bash
# Tentatives de connexion SSH échouées
grep 'Failed password' /var/log/auth.log | \
  awk '{print $(NF-3)}' | sort | uniq -c | sort -rn | head -20

# Connexions réussies
grep 'Accepted' /var/log/auth.log | \
  awk '{print $9, $11}' | sort | uniq -c

# Avec journald
journalctl -u ssh --since "24h ago" | \
  grep -E 'Failed|Invalid user|Accepted'

# Analyse avec logwatch (si installé)
logwatch --service sshd --range today --detail high

# Detect les scans de ports dans les logs (connexions SYN sans suite)
grep 'Invalid user' /var/log/auth.log | \
  grep -oE '[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}' | \
  sort | uniq -c | sort -rn | head
```

## OSSEC / Wazuh — HIDS

```bash
# Wazuh agent installation (RHEL/CentOS)
curl -s https://packages.wazuh.com/key/GPG-KEY-WAZUH | gpg --dearmor > \
  /usr/share/keyrings/wazuh.gpg
echo "deb [signed-by=/usr/share/keyrings/wazuh.gpg] \
https://packages.wazuh.com/4.x/apt/ stable main" > \
  /etc/apt/sources.list.d/wazuh.list
apt update && apt install wazuh-agent -y

# Configurer le manager
MGR_IP=10.0.0.1
wazuh-agentd -m $MGR_IP

# Vérifier l'état
/var/ossec/bin/wazuh-control status

# Lancer un check d'intégrité fichiers
/var/ossec/bin/wazuh-agentd -t
```
