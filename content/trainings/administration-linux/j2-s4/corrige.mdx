# Corrigé — Détection d'intrusion

## Exercice 1

```bash
# Capture 5 minutes
IF=$(ip route | grep default | awk '{print $5}' | head -1)
tcpdump -i $IF -w /tmp/traffic.pcap &
TDP=$!
sleep 300
kill $TDP

# Top 10 couples source:destination
tcpdump -r /tmp/traffic.pcap -nn -q 2>/dev/null | \
  awk '{print $3, $5}' | sort | uniq -c | sort -rn | head -10

# Tentatives SSH échouées
tcpdump -r /tmp/traffic.pcap -nn 'port 22 and tcp[tcpflags] & tcp-rst != 0' 2>/dev/null | head -20

# Requêtes DNS
tcpdump -r /tmp/traffic.pcap -nn 'port 53' 2>/dev/null | \
  grep -oE ' [A-Za-z0-9._-]+\.' | sort | uniq -c | sort -rn | head -20
```

## Exercice 2 — Suricata

```bash
apt install suricata -y
suricata-update

# Règle personnalisée anti-nmap
cat > /etc/suricata/rules/local.rules << 'EOF'
# Détection User-Agent Nmap
alert http any any -> any any (
  msg:"SCAN Nmap HTTP User-Agent détecté";
  flow:to_server,established;
  http.user_agent; content:"Nmap";
  classtype:network-scan;
  sid:9000001;
  rev:1;
)

# Détection SYN scan rapide
alert tcp any any -> $HOME_NET any (
  msg:"SCAN SYN scan détecté";
  flags:S;
  flow:stateless;
  threshold: type threshold, track by_src, count 20, seconds 1;
  classtype:network-scan;
  sid:9000002;
  rev:1;
)
EOF

# Ajouter la règle locale dans la config
echo 'rule-files:' >> /etc/suricata/suricata.yaml
echo '  - /etc/suricata/rules/local.rules' >> /etc/suricata/suricata.yaml

suricata -T -c /etc/suricata/suricata.yaml  # Test syntaxe
systemctl restart suricata

# Test
nmap -sS 127.0.0.1 &
jq 'select(.event_type=="alert") | .alert.signature' /var/log/suricata/eve.json | tail -5
```

## Exercice 3

```bash
# Top 10 IPs brute-force SSH (7 jours)
journal_args="-u ssh --since '7 days ago'"
journalctl -u ssh --since "7 days ago" | \
  grep 'Failed password' | \
  grep -oE '[0-9]{1,3}(\.[0-9]{1,3}){3}' | \
  sort | uniq -c | sort -rn | head -10

# Connexions hors heures ouvrables
journalctl -u ssh | grep 'Accepted' | \
  awk '{h=substr($3,1,2); if(h<8 || h>19) print}' | head -20

# Rapport HTML maison
cat > /tmp/auth-report.html << 'HTML'
```

## Exercice 4 — Règles avancées

```bash
cat >> /etc/suricata/rules/local.rules << 'EOF'
# LFI /etc/passwd
alert http any any -> $HTTP_SERVERS any (
  msg:"WEB LFI tentative /etc/passwd";
  flow:to_server,established;
  http.uri; content:"/etc/passwd";
  classtype:web-application-attack;
  sid:9000003;
  rev:1;
)

# ICMP flood
alert icmp any any -> any any (
  msg:"ICMP flood détecté";
  threshold: type both, track by_dst, count 100, seconds 60;
  classtype:bad-unknown;
  sid:9000004;
  rev:1;
)

# SSH hors port 22
alert tcp any any -> $HOME_NET !22 (
  msg:"SSH hors port standard";
  flow:to_server;
  app-layer-protocol:ssh;
  classtype:policy-violation;
  sid:9000005;
  rev:1;
)
EOF

suricata -T -c /etc/suricata/suricata.yaml && systemctl reload suricata
```
