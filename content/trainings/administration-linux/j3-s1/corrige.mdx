# Corrigé — Cryptographie appliquée

## Exercice 1

```bash
# Chiffrement
openssl enc -aes-256-gcm -pbkdf2 -iter 200000 \
  -in /etc/passwd -out /tmp/passwd.enc -k "P@ssw0rd_F0rmat10n!"

# Vérification : fichier opaque
head -c 50 /tmp/passwd.enc | strings

# Déchiffrement
openssl enc -d -aes-256-gcm -pbkdf2 -iter 200000 \
  -in /tmp/passwd.enc -out /tmp/passwd.dec -k "P@ssw0rd_F0rmat10n!"

# Vérification intégrité
sha256sum /etc/passwd /tmp/passwd.dec

# Benchmark
dd if=/dev/zero bs=1M count=1024 | openssl enc -aes-128-cbc -pass pass:test > /dev/null
dd if=/dev/zero bs=1M count=1024 | openssl enc -aes-256-gcm -pass pass:test > /dev/null
# Ou :
openssl speed aes-128-cbc aes-256-gcm
```

## Exercice 2

```bash
# Générer la paire RSA
openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:4096 -out /tmp/private.pem
openssl rsa -in /tmp/private.pem -pubout -out /tmp/public.pem

# Document à signer
echo "Formation Sécurité Linux - Acte authentique" > /tmp/document.txt

# Signature
openssl dgst -sha256 -sign /tmp/private.pem \
  -out /tmp/document.sig /tmp/document.txt

# Vérification (OK)
openssl dgst -sha256 -verify /tmp/public.pem \
  -signature /tmp/document.sig /tmp/document.txt
# Output : Verified OK

# Après modification
echo " (modifié)" >> /tmp/document.txt
openssl dgst -sha256 -verify /tmp/public.pem \
  -signature /tmp/document.sig /tmp/document.txt
# Output : Verification Failure
```

## Exercice 3

```bash
# Certificat Gmail
openssl s_client -connect mail.google.com:443 </dev/null 2>/dev/null | \
  openssl x509 -noout -text | grep -E 'Issuer|Not|Subject:|Signature Algorithm|Public Key'

# Suites cryptographiques (avec nmap)
nmap --script ssl-enum-ciphers -p 443 mail.google.com

# Détection RC4/DES/3DES
nmap --script ssl-enum-ciphers -p 443 mail.google.com | \
  grep -iE 'rc4|des|3des|null|anon'
```

## Exercice 4

```bash
# Manifest SHA-256
sha256sum /usr/bin/* 2>/dev/null > /root/manifest-usrbin.sha256
chmod 600 /root/manifest-usrbin.sha256

# Simulation de modification
cp /usr/bin/which /usr/bin/which.bak
echo "# compromis" >> /usr/bin/which

# Détection
sha256sum -c /root/manifest-usrbin.sha256 2>/dev/null | grep FAILED

# Restauration
mv /usr/bin/which.bak /usr/bin/which

# Pourquoi pas MD5 :
# Collisions connues (Wang & Yu, 2004)
# Création de deux fichiers différents avec même MD5 en <1s
openssl speed md5 sha256  # MD5 n'est pas plus rapide pour justifier son usage
echo "MD5 : susceptible aux attaques par collision, ne garantit pas l'intégrité"
```
